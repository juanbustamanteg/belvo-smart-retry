## 4. Diagnostic method

The rule definition belongs to master section 5. For traceability, this execution used the following method fingerprint:

1. Read the original portfolio and all requests, preserving identifiers and money values.
2. Group by loan and parsed creation instant; retain all latest-batch members.
3. Derive timestamp validity, chargeback/success flags, mixed outcomes/reasons, repeated amounts and homogeneous code-04 failure.
4. Anchor cooldown to the latest member's later creation/completion time, converted to the local calendar date.
5. Apply the master's ordered reason selection with 3, 7 and 14 calendar days separately.
6. Left-join all 3,062 loans and reconcile counts. No recommended amount is calculated.
7. Compare per-loan Python decisions with SQLite decisions, repeat under reversed row order, exercise constructed branch cases and verify unchanged source hashes.

The temporary utility calls the review outcome `REVIEW`; it maps **one-to-one to canonical `REVIEW_REMEDIATE`** and is not another product decision. The numeric source reason `4` corresponds to normalized `04`. The diagnostic's CSV parser and SQL staging are not a complete tested implementation of future malformed-file handling or every code-normalization case.

This is retrospective analysis, not a historical point-in-time backtest. Final outcomes, current debt and the chosen cutoff do not reconstruct prior event availability, delinquency episodes or current consent/method state.

## Appendix A. Exact executed baseline utility

The code below is preserved as diagnostic evidence, not an approved app architecture.

```python
import csv, datetime as d, collections, sqlite3, json, hashlib
from decimal import Decimal
from pathlib import Path
from zoneinfo import ZoneInfo

root=Path('<project folder>')
tz=ZoneInfo('America/Mexico_City')
cut=d.datetime(2025,11,11,23,59,59,tzinfo=tz)
def parse(s): return d.datetime.fromisoformat(s.replace('Z','+00:00')) if s else None
loans=list(csv.DictReader((root/'PaymentsPM_Challenge_loan_info.csv').open()))
rows=list(csv.DictReader((root/'PaymentsPM_Challenge_payment_requests.csv').open()))
assert len(loans)==3062 and len(rows)==253006
hist=collections.defaultdict(list)
db=sqlite3.connect(':memory:')
db.execute('CREATE TABLE loan(loan_id TEXT PRIMARY KEY, balance REAL)')
db.executemany('INSERT INTO loan VALUES (?,?)',[(r['loan_id'],float(r['total_amount_outstanding'])) for r in loans])
db.execute('CREATE TABLE p(loan_id TEXT, ts REAL, amount TEXT, status TEXT, reason TEXT, anchor_day INTEGER, invalid INTEGER)')
for r in rows:
    t=parse(r['created_at']); c=parse(r['completed_at']); cb=parse(r['chargeback_at'])
    bad=int(c is None or c<t or c>cut or (cb is not None and (cb<t or (c is not None and cb<c) or cb>cut)))
    a=max(t,c or t).astimezone(tz).date().toordinal()
    r.update(ts=t.timestamp(),anchor=a,invalid=bad)
    hist[r['loan_id']].append(r)
    db.execute('INSERT INTO p VALUES (?,?,?,?,?,?,?)',(r['loan_id'],t.timestamp(),str(Decimal(r['amount'])),r['status'],r['failed_reason'],a,bad))

def features(rr):
    if not rr:return dict(n=0)
    last=max(r['ts'] for r in rr); b=[r for r in rr if r['ts']==last]
    return dict(n=len(b),invalid=any(r['invalid'] for r in b),cb=any(r['status']=='chargeback' for r in b),
      dup=len({Decimal(r['amount']) for r in b})<len(b),mixed=len({r['status'] for r in b})>1 or len({r['failed_reason'] for r in b if r['status']=='failed'})>1,
      success=any(r['status']=='successful' for r in b),soft=all(r['status']=='failed' and r['failed_reason']=='4' for r in b),
      anchor=max(r['anchor'] for r in b),last=last)
def classify(f,k,balance):
    if balance<=0:return 'REVIEW','INVALID_BALANCE'
    if not f['n']:return 'REVIEW','NO_HISTORY'
    if f['invalid']:return 'REVIEW','INVALID_TIMESTAMP'
    if f['cb']:return 'REVIEW','LATEST_CHARGEBACK'
    if f['mixed'] or f['dup']:return 'REVIEW','AMBIGUOUS_LATEST_BATCH'
    if f['success']:return 'REVIEW','LATEST_SUCCESS_RECONCILE'
    if not f['soft']:return 'REVIEW','ACCOUNT_AUTH_OR_UNKNOWN_REASON'
    if cut.date().toordinal()<f['anchor']+k:return 'WAIT','COOLDOWN'
    return 'RETRY','SOFT_FAILURE_WINDOW_ELAPSED'

sql='''WITH latest AS (SELECT loan_id,MAX(ts) AS ts FROM p GROUP BY loan_id),
b AS (SELECT p.* FROM p JOIN latest l ON p.loan_id=l.loan_id AND p.ts=l.ts),
f AS (SELECT loan_id,COUNT(*) AS n, MAX(invalid) AS invalid,
 MAX(CASE WHEN status='chargeback' THEN 1 ELSE 0 END) AS cb,
 CASE WHEN COUNT(DISTINCT amount)<COUNT(*) THEN 1 ELSE 0 END AS dup,
 CASE WHEN COUNT(DISTINCT status)>1 OR COUNT(DISTINCT CASE WHEN status='failed' THEN reason END)>1 THEN 1 ELSE 0 END AS mixed,
 MAX(CASE WHEN status='successful' THEN 1 ELSE 0 END) AS success,
 MIN(CASE WHEN status='failed' AND reason='4' THEN 1 ELSE 0 END) AS soft, MAX(anchor_day) AS anchor
 FROM b GROUP BY loan_id)
SELECT l.loan_id,CASE WHEN l.balance<=0 OR f.n IS NULL OR f.invalid=1 OR f.cb=1 OR f.dup=1 OR f.mixed=1 OR f.success=1 OR f.soft<>1 THEN 'REVIEW'
 WHEN ? < f.anchor + ? THEN 'WAIT' ELSE 'RETRY' END AS decision
FROM loan l LEFT JOIN f ON l.loan_id=f.loan_id'''
out={}
for k in [3,7,14]:
    py={l['loan_id']:classify(features(hist[l['loan_id']]),k,Decimal(l['total_amount_outstanding'])) for l in loans}
    sq=dict(db.execute(sql,(cut.date().toordinal(),k)))
    assert {i:v[0] for i,v in py.items()}==sq
    totals=collections.Counter(v[0] for v in py.values());reasons=collections.Counter(v[1] for v in py.values())
    assert sum(totals.values())==3062
    examples=[]
    for outcome in ['RETRY','WAIT','REVIEW']:
        ids=sorted(i for i,v in py.items() if v[0]==outcome)
        for i in ids[:1]:
            f=features(hist[i]);examples.append(dict(loan_id=i,decision=outcome,reason=py[i][1],latest_rows=f.get('n'),anchor_date=d.date.fromordinal(f['anchor']).isoformat() if 'anchor' in f else None))
    out[k]=dict(counts=dict(totals),reasons=dict(reasons),examples=examples)
assert classify(dict(n=1,invalid=False,cb=False,dup=False,mixed=False,success=False,soft=True,anchor=cut.date().toordinal()-7),7,Decimal(100))[0]=='RETRY'
assert classify(dict(n=1,invalid=False,cb=False,dup=False,mixed=False,success=False,soft=True,anchor=cut.date().toordinal()-6),7,Decimal(100))[0]=='WAIT'
print(json.dumps({'cutoff':cut.isoformat(),'results':out,'sqlite_python_match':True,'source_hashes':{f:hashlib.sha256((root/f).read_bytes()).hexdigest() for f in ['PaymentsPM_Challenge_loan_info.csv','PaymentsPM_Challenge_payment_requests.csv']}},indent=2))
```
